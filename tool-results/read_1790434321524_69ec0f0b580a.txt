    49→const num = (v: string): number => {
    50→  const n = parseFloat(v);
    51→  return Number.isFinite(n) ? n : 0;
    52→};
    53→const round2 = (n: number) => Math.round(n * 100) / 100;
    54→
    55→/** Mirrors the server formula: total = subtotal - discount + subtotal * tax% / 100 */
    56→function previewTotals(items: ItemDraft[], discount: number, taxPercent: number) {
    57→  const subtotal = round2(items.reduce((s, i) => s + num(i.quantity) * num(i.unitPrice), 0));
    58→  const tax = round2((subtotal * taxPercent) / 100);
    59→  return { subtotal, tax, total: round2(subtotal - discount + tax) };
    60→}

    62→function extractList(data: unknown): Record<string, unknown>[] {
    63→  if (Array.isArray(data)) return data as Record<string, unknown>[];
    64→  if (data && typeof data === "object") {
    65→    const obj = data as Record<string, unknown>;
    66→    for (const key of ["items", "leads", "clients", "projects", "team"]) {
    67→      if (Array.isArray(obj[key])) return obj[key] as Record<string, unknown>[];
    68→    }
    69→  }
    70→  return [];
    71→}
    72→
    73→function toOptions(url: string, labelKey: string): Promise<Option[]> {
    74→  return api.get<unknown>(url)
    75→    .then((data) => extractList(data).map((x) => ({ id: String(x.id), label: String(x[labelKey] ?? x.name ?? x.companyName ?? "") })).filter((o) => o.label))
    76→    .catch(() => []);
    77→}

    79→function toDateInput(d: string | Date): string {
    80→  const x = new Date(d);
    81→  const p = (n: number) => String(n).padStart(2, "0");
    82→  return `${x.getFullYear()}-${p(x.getMonth() + 1)}-${p(x.getDate())}`;
    83→}

   430→      setBuilderOpen(true);
   431→    } catch (e) {
   432→      toast({ title: "Could not load quotation", description: e instanceof Error ? e.message : undefined });
   433→    }
   434→  };
   435→
   436→  const submitBuilder = async () => {
   437→    if (form.title.trim().length < 2) return toast({ title: "Title is required" });
   438→    const cleaned = items.filter((i) => i.description.trim());
   439→    if (cleaned.length === 0) return toast({ title: "Add at least one item with a description" });
   440→    for (const i of cleaned) {
   441→      if (num(i.quantity) <= 0) return toast({ title: `Quantity for "${i.description}" must be greater than 0` });
   442→      if (num(i.unitPrice) < 0) return toast({ title: `Unit price for "${i.description}" cannot be negative` });
   443→    }
   444→    setSaving(true);
   445→    try {
   446→      const payload = {
   447→        title: form.title.trim(),
   448→        leadId: form.leadId || null,
   449→        clientId: form.clientId || null,
   450→        validUntil: form.validUntil || null,
   451→        currency: form.currency,
   452→        discountAmount: num(form.discountAmount),
   453→        taxPercent: num(form.taxPercent),
   454→        paymentTerms: form.paymentTerms || null,
   455→        notes: form.notes || null,
   456→        items: cleaned.map((i) => ({ description: i.description.trim(), quantity: num(i.quantity), unitPrice: num(i.unitPrice) })),
   457→      };
   458→      if (editingId) {
   459→        await api.patch(`/api/quotations/${editingId}`, payload);
   460→        toast({ title: "Quotation updated" });
   461→      } else {
   462→        await api.post("/api/quotations", payload);
   463→        toast({ title: "Quotation created as draft" });
   464→      }
   465→      setBuilderOpen(false);
   466→      load();
   467→    } catch (e) {
   468→      toast({ title: "Save failed", description: e instanceof Error ? e.message : undefined });
   469→    } finally {
   470→      setSaving(false);
   471→    }
   472→  };
   473→
   474→  const confirmDelete = async () => {
   475→    if (!deleting) return;
   476→    try {
   477→      await api.delete(`/api/quotations/${deleting.id}`);
   478→      toast({ title: "Draft deleted" });
   479→      setDeleting(null);
   480→      if (detailId === deleting.id) setSheetOpen(false);
   481→      load();
   482→    } catch (e) {
   483→      toast({ title: "Delete failed", description: e instanceof Error ? e.message : undefined });
   484→    }
   485→  };
   486→
   487→  const totals = previewTotals(items, num(form.discountAmount), num(form.taxPercent));
   488→  const totalPages = Math.max(1, Math.ceil(total / pageSize));
   489→
   490→  return (
   491→    <div>
   492→      <PageHeader
   493→        title="Quotations"
   494→        description="Quick priced quotes with validity dates — the lightweight sibling of proposals."
   495→        actions={canCreate && <Button onClick={openCreate}><Plus className="w-4 h-4 mr-2" /> New Quotation</Button>}
   496→      />
   497→
   498→      <div className="flex flex-col sm:flex-row gap-2 mb-4">
   499→        <Input placeholder="Search number or title…" value={search} onChange={(e) => onSearchInput(e.target.value)} className="sm:max-w-xs" />
   500→        <Select value={statusFilter} onValueChange={setStatusFilter}>
   501→          <SelectTrigger className="sm:w-44"><SelectValue /></SelectTrigger>
   502→          <SelectContent>
   503→            <SelectItem value="ALL">All statuses</SelectItem>
   504→            <SelectItem value="DRAFT">Draft</SelectItem>
   505→            <SelectItem value="SENT">Sent</SelectItem>
   506→            <SelectItem value="ACCEPTED">Accepted</SelectItem>
   507→            <SelectItem value="REJECTED">Rejected</SelectItem>
   508→            <SelectItem value="EXPIRED">Expired</SelectItem>
   509→          </SelectContent>
   510→        </Select>
   511→      </div>
   512→
   513→      {loading ? (
   514→        <ListSkeleton />
   515→      ) : error ? (
   516→        <ErrorState onRetry={() => load()} />
   517→      ) : total === 0 ? (
   518→        <EmptyState
   519→          icon={<ReceiptText className="w-5 h-5" />}
   520→          title={query || statusFilter !== "ALL" ? "No quotations match your filters" : "No quotations yet"}
   521→          description={query || statusFilter !== "ALL" ? "Try adjusting the search or status filter." : "Send your first quick quote with priced items."}
   522→          action={canCreate && !(query || statusFilter !== "ALL") ? <Button size="sm" onClick={openCreate}><Plus className="w-4 h-4 mr-2" /> New Quotation</Button> : undefined}
   523→        />
   524→      ) : (
   525→        <div className="apex-panel overflow-x-auto">
   526→          <table className="w-full text-sm">
   527→            <thead>
   528→              <tr className="border-b border-border text-left text-[11px] uppercase tracking-wide text-muted-foreground">
   529→                <th className="px-4 py-3 font-medium">Quotation #</th>
   530→                <th className="px-4 py-3 font-medium">Title</th>
   531→                <th className="px-4 py-3 font-medium hidden lg:table-cell">Lead / Client</th>
   532→                <th className="px-4 py-3 font-medium text-right">Total</th>
   533→                <th className="px-4 py-3 font-medium hidden md:table-cell">Status</th>
   534→                <th className="px-4 py-3 font-medium hidden xl:table-cell">Valid until</th>
   535→                <th className="px-4 py-3 font-medium text-right">Actions</th>
   536→              </tr>
   537→            </thead>
   538→            <tbody className="divide-y divide-border">
   539→              {rows.map((r) => {
   540→                const expired = r.validUntil ? new Date(r.validUntil) < new Date() && ACTIVE_STATUSES.includes(r.status) : false;
   541→                return (
   542→                  <tr key={r.id} className="hover:bg-accent/40 transition-colors">
   543→                    <td className="px-4 py-3 font-mono text-xs whitespace-nowrap">{r.quotationNumber}</td>
   544→                    <td className="px-4 py-3">
   545→                      <button className="text-left font-medium hover:text-primary transition-colors" onClick={() => { setDetailId(r.id); setSheetOpen(true); }}>
   546→                        {r.title}
   547→                      </button>
   548→                      <p className="text-[11px] text-muted-foreground mt-0.5 lg:hidden">{r.lead?.companyName ?? r.client?.companyName ?? "Unlinked"}</p>
   549→                    </td>
   550→                    <td className="px-4 py-3 hidden lg:table-cell text-muted-foreground">{r.lead?.companyName ?? r.client?.companyName ?? "Unlinked"}</td>
   551→                    <td className="px-4 py-3 text-right font-medium tabular-nums whitespace-nowrap">{formatCurrency(r.total, r.currency)}</td>
   552→                    <td className="px-4 py-3 hidden md:table-cell"><StatusBadge status={r.status} /></td>
   553→                    <td className={cn("px-4 py-3 hidden xl:table-cell whitespace-nowrap", expired ? "text-rose-400" : "text-muted-foreground")}>
   554→                      {formatDate(r.validUntil)}{expired && " · expired"}
   555→                    </td>
   556→                    <td className="px-4 py-3 text-right">
   557→                      <div className="inline-flex items-center gap-1">
   558→                        <Button variant="ghost" size="icon" className="h-8 w-8" title="View" onClick={() => { setDetailId(r.id); setSheetOpen(true); }}>
   559→                          <Eye className="w-3.5 h-3.5" />
   560→                        </Button>
   561→                        {canEdit && !["ACCEPTED", "REJECTED", "EXPIRED"].includes(r.status) && (
   562→                          <Button variant="ghost" size="icon" className="h-8 w-8" title="Edit" onClick={() => openEdit(r)}>
   563→                            <Pencil className="w-3.5 h-3.5" />
   564→                          </Button>
   565→                        )}
   566→                        {canDelete && r.status === "DRAFT" && (
   567→                          <Button variant="ghost" size="icon" className="h-8 w-8 text-muted-foreground hover:text-destructive" title="Delete draft" onClick={() => setDeleting(r)}>
   568→                            <Trash2 className="w-3.5 h-3.5" />
   569→                          </Button>
   570→                        )}
   571→                        {!canEdit && !canDelete && (
   572→                          <DropdownMenu>
   573→                            <DropdownMenuTrigger asChild>
   574→                              <Button variant="ghost" size="icon" className="h-8 w-8"><MoreHorizontal className="w-4 h-4" /></Button>
   575→                            </DropdownMenuTrigger>
   576→                            <DropdownMenuContent align="end"><DropdownMenuItem disabled>No actions available</DropdownMenuItem></DropdownMenuContent>
   577→                          </DropdownMenu>
   578→                        )}
   579→                      </div>
   580→                    </td>
   581→                  </tr>
   582→                );
   583→              })}
   584→            </tbody>
   585→          </table>
   586→        </div>
   587→      )}
   588→
   589→      {totalPages > 1 && (
   590→        <div className="flex items-center justify-between text-sm mt-4">
   591→          <p className="text-muted-foreground">Page {page} of {totalPages} · {total} quotations</p>
   592→          <div className="flex gap-2">
   593→            <Button variant="outline" size="sm" disabled={page <= 1} onClick={() => load({ page: page - 1 })}>Previous</Button>
   594→            <Button variant="outline" size="sm" disabled={page >= totalPages} onClick={() => load({ page: page + 1 })}>Next</Button>
   595→          </div>
   596→        </div>
   597→      )}
   598→
   599→      {/* Builder dialog */}
   600→      <Dialog open={builderOpen} onOpenChange={setBuilderOpen}>
   601→        <DialogContent className="sm:max-w-2xl">
   602→          <DialogHeader>
   603→            <DialogTitle>{editingId ? "Edit quotation" : "New quotation"}</DialogTitle>
   604→            <DialogDescription>Totals are always recalculated on the server from the line items below.</DialogDescription>
   605→          </DialogHeader>
   606→          <div className="max-h-[70vh] overflow-y-auto apex-scroll pr-1 space-y-4">
   607→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
   608→              <Field label="Title" required>
   609→                <Input value={form.title} onChange={(e) => setForm({ ...form, title: e.target.value })} placeholder="e.g. Social media retainer" />
   610→              </Field>
   611→              <Field label="Currency">
   612→                <Select value={form.currency} onValueChange={(v) => setForm({ ...form, currency: v })}>
   613→                  <SelectTrigger><SelectValue /></SelectTrigger>
   614→                  <SelectContent>
   615→                    <SelectItem value="EGP">EGP</SelectItem>
   616→                    <SelectItem value="USD">USD</SelectItem>
   617→                    <SelectItem value="SAR">SAR</SelectItem>
   618→                  </SelectContent>
   619→                </Select>
   620→              </Field>
   621→              <Field label="Valid until">
   622→                <Input type="date" value={form.validUntil} onChange={(e) => setForm({ ...form, validUntil: e.target.value })} />
   623→              </Field>
   624→            </div>
   625→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   626→              {leadOpts.length > 0 ? (
   627→                <Field label="Linked lead">
   628→                  <Select value={form.leadId || "NONE"} onValueChange={(v) => setForm({ ...form, leadId: v === "NONE" ? "" : v, clientId: "" })}>
   629→                    <SelectTrigger><SelectValue placeholder="None" /></SelectTrigger>
   630→                    <SelectContent className="max-h-60">
   631→                      <SelectItem value="NONE">None</SelectItem>
   632→                      {leadOpts.map((o) => <SelectItem key={o.id} value={o.id}>{o.label}</SelectItem>)}
   633→                    </SelectContent>
   634→                  </Select>
   635→                </Field>
   636→              ) : (
   637→                <Field label="Linked lead" hint="Lead list unavailable — the quotation will be unlinked">
   638→                  <Input value={form.leadId} disabled placeholder="Unavailable without leads.view" />
   639→                </Field>
   640→              )}
   641→              {clientOpts.length > 0 ? (
   642→                <Field label="Linked client">
   643→                  <Select value={form.clientId || "NONE"} onValueChange={(v) => setForm({ ...form, clientId: v === "NONE" ? "" : v, leadId: "" })}>
   644→                    <SelectTrigger><SelectValue placeholder="None" /></SelectTrigger>
   645→                    <SelectContent className="max-h-60">
   646→                      <SelectItem value="NONE">None</SelectItem>
   647→                      {clientOpts.map((o) => <SelectItem key={o.id} value={o.id}>{o.label}</SelectItem>)}
   648→                    </SelectContent>
   649→                  </Select>
   650→                </Field>
   651→              ) : (
   652→                <Field label="Linked client" hint="Client list unavailable — the quotation will be unlinked">
   653→                  <Input value={form.clientId} disabled placeholder="Unavailable without clients.view" />
   654→                </Field>
   655→              )}
   656→            </div>
   657→
   658→            <div>
   659→              <p className="text-xs font-medium text-muted-foreground mb-2">Line items</p>
   660→              <ItemsEditor items={items} onChange={setItems} currency={form.currency} />
   661→              <div className="mt-3 rounded-lg border border-border bg-secondary/30 p-4">
   662→                <div className="grid grid-cols-2 gap-3">
   663→                  <Field label="Discount amount">
   664→                    <Input type="number" min={0} step={0.01} value={form.discountAmount} onChange={(e) => setForm({ ...form, discountAmount: e.target.value })} />
   665→                  </Field>
   666→                  <Field label="Tax percent">
   667→                    <Input type="number" min={0} max={100} step={0.01} value={form.taxPercent} onChange={(e) => setForm({ ...form, taxPercent: e.target.value })} />
   668→                  </Field>
   669→                </div>
   670→                <Separator className="my-3" />
   671→                <div className="space-y-1.5 text-sm">
   672→                  <div className="flex justify-between"><span className="text-muted-foreground">Subtotal</span><span className="tabular-nums">{formatCurrency(totals.subtotal, form.currency)}</span></div>
   673→                  <div className="flex justify-between"><span className="text-muted-foreground">Tax ({num(form.taxPercent)}%)</span><span className="tabular-nums">{formatCurrency(totals.tax, form.currency)}</span></div>
   674→                  <div className="flex justify-between font-semibold"><span>Total</span><span className="tabular-nums text-primary">{formatCurrency(totals.total, form.currency)}</span></div>
   675→                </div>
   676→              </div>
   677→            </div>
   678→
   679→            <Field label="Payment terms">
   680→              <Textarea rows={2} value={form.paymentTerms} onChange={(e) => setForm({ ...form, paymentTerms: e.target.value })} placeholder="e.g. Full payment on delivery" />
   681→            </Field>
   682→            <Field label="Internal notes">
   683→              <Textarea rows={2} value={form.notes} onChange={(e) => setForm({ ...form, notes: e.target.value })} placeholder="Notes the client will never see" />
   684→            </Field>
   685→          </div>
   686→          <DialogFooter>
   687→            <Button variant="outline" onClick={() => setBuilderOpen(false)}>Cancel</Button>
   688→            <Button onClick={submitBuilder} disabled={saving}>
   689→              {saving ? <RefreshCw className="w-4 h-4 mr-2 animate-spin" /> : null}
   690→              {editingId ? "Save changes" : "Create quotation"}
   691→            </Button>
   692→          </DialogFooter>
   693→        </DialogContent>
   694→      </Dialog>
   695→
   696→      {/* Detail sheet */}
   697→      <QuotationDetailSheet
   698→        id={detailId}
   699→        open={sheetOpen}
   700→        onOpenChange={setSheetOpen}
   701→        onChanged={() => load()}
   702→        canEdit={canEdit}
   703→        canDelete={canDelete}
   704→        onDeleteRequest={(row) => setDeleting(row)}
   705→      />
   706→
   707→      {/* Delete confirm */}
   708→      <AlertDialog open={!!deleting} onOpenChange={(o) => !o && setDeleting(null)}>
   709→        <AlertDialogContent>
   710→          <AlertDialogHeader>
   711→            <AlertDialogTitle>Delete this draft?</AlertDialogTitle>
   712→            <AlertDialogDescription>
   713→              {deleting ? `${deleting.quotationNumber} — ${deleting.title}` : ""} will be permanently removed. Only drafts can be deleted.
   714→            </AlertDialogDescription>
   715→          </AlertDialogHeader>
   716→          <AlertDialogFooter>
   717→            <AlertDialogCancel>Keep draft</AlertDialogCancel>
   718→            <AlertDialogAction className="bg-destructive text-destructive-foreground hover:bg-destructive/90" onClick={confirmDelete}>Delete</AlertDialogAction>
   719→          </AlertDialogFooter>
   720→        </AlertDialogContent>
   721→      </AlertDialog>
   722→    </div>
   723→  );
   724→}
