   134→function deadlineMeta(deadline?: string | null, status?: string): { text: string; cls: string } {
   135→  if (!deadline) return { text: "No deadline", cls: "text-muted-foreground" };
   136→  const d = new Date(deadline);
   137→  d.setHours(0, 0, 0, 0);
   138→  const today = new Date();
   139→  today.setHours(0, 0, 0, 0);
   140→  const days = Math.round((d.getTime() - today.getTime()) / 86400000);
   141→  const urgent = !!status && !["COMPLETED", "CANCELLED"].includes(status) && days < 7;
   142→  return { text: formatDate(deadline), cls: urgent ? "text-rose-300 font-medium" : "text-muted-foreground" };
   143→}

   149→function extractClients(data: unknown): ClientOption[] {
   150→  if (Array.isArray(data)) return data as ClientOption[];
   151→  if (data && typeof data === "object") {
   152→    const obj = data as Record<string, unknown>;
   153→    if (Array.isArray(obj.items)) return obj.items as ClientOption[];
   154→    if (Array.isArray(obj.clients)) return obj.clients as ClientOption[];
   155→  }
   156→  return [];
   157→}

   177→  const [type, setType] = useState("BUSINESS_WEBSITE");

   179→  const [teamUnavailable, setTeamUnavailable] = useState(false);

   186→
   187→  // Load dialog data (degrade gracefully per source)
   188→  useEffect(() => {
   189→    if (!open) return;
   190→    setName(""); setDescription(""); setClientId("none"); setManualClientId("");
   191→    setType("BUSINESS_WEBSITE"); setManagerId("none"); setBudget("");

   207→          setTypes(projectsSetting.types);

   210→      .catch(() => setTypes(FALLBACK_TYPES)); // no settings.manage → hardcoded defaults
   211→
   212→    api.get<{ team: TeamMemberOption[] }>("/api/team")
   213→      .then((d) => setTeam(d.team))
   214→      .catch(() => setTeamUnavailable(true));

   218→    const resolvedClientId = clientUnavailable ? manualClientId.trim() : clientId === "none" ? "" : clientId;

   223→        name: name.trim(),
   224→        clientId: resolvedClientId,

   238→        title: "Could not create project",
   239→        description: e instanceof ApiClientError ? e.message : undefined,
   240→        variant: "destructive",
   241→      });
   242→    } finally {
   243→      setSaving(false);
   244→    }
   245→  };
   246→
   247→  return (
   248→    <Dialog open={open} onOpenChange={onOpenChange}>
   249→      <DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto apex-scroll">

   257→

   260→              label="Client ID" required
   261→              hint="Client list is unavailable (missing clients.view or the CRM module is not ready). Paste a Client ID from the CRM module."
   262→            >
   263→              <Input value={manualClientId} onChange={(e) => setManualClientId(e.target.value)} placeholder="e.g. cm5x… (APX-CLT-0001 ID)" />
   264→            </Field>

   266→            <Field label="Client" required>
   267→              <Select value={clientId} onValueChange={setClientId}>
   268→                <SelectTrigger className="h-9"><SelectValue placeholder="Select client" /></SelectTrigger>

   280→            <Select value={type} onValueChange={setType}>
   281→              <SelectTrigger className="h-9"><SelectValue /></SelectTrigger>
   282→              <SelectContent>
   283→                {types.map((t) => <SelectItem key={t.key} value={t.key}>{t.label}</SelectItem>)}
   284→              </SelectContent>
   285→            </Select>
   286→          </Field>
   287→
   288→          <Field label="Manager" hint={teamUnavailable ? "Team list unavailable — manager can be assigned later." : undefined}>
   289→            <Select value={managerId} onValueChange={setManagerId} disabled={teamUnavailable}>
   290→              <SelectTrigger className="h-9"><SelectValue placeholder="No manager" /></SelectTrigger>
   291→              <SelectContent>
   292→                <SelectItem value="none">No manager</SelectItem>

   338→function QuickAddTask({
   339→  projectId, phaseId, onAdded, canCreate,
   340→}: {
   341→  projectId: string;
   342→  phaseId: string;
   343→  onAdded: () => void;
   344→  canCreate: boolean;
   345→}) {
   346→  const { toast } = useToast();
   347→  const [title, setTitle] = useState("");
   348→  const [busy, setBusy] = useState(false);
   349→
   350→  if (!canCreate) return null;
   351→
   352→  const add = async () => {
   353→    if (!title.trim()) return;
   354→    setBusy(true);
   355→    try {
   356→      await api.post("/api/tasks", { title: title.trim(), projectId, phaseId });
   357→      setTitle("");
   358→      toast({ title: "Task added to phase" });
   359→      onAdded();
   360→    } catch (e) {
   361→      toast({
   362→        title: "Could not add task",
   363→        description: e instanceof ApiClientError ? e.message : undefined,
   364→        variant: "destructive",
   365→      });
   366→    } finally {
   367→      setBusy(false);
   368→    }
   369→  };
   370→
   371→  return (
   372→    <div className="flex gap-1.5 mt-2">
   373→      <Input
   374→        value={title}
   375→        onChange={(e) => setTitle(e.target.value)}
   376→        onKeyDown={(e) => { if (e.key === "Enter") { e.preventDefault(); void add(); } }}
   377→        placeholder="Quick add task…"
   378→        className="h-8 text-xs"
   379→      />
   380→      <Button size="sm" variant="outline" className="h-8 px-2.5" disabled={!title.trim() || busy} onClick={() => void add()} aria-label="Add task">

   382→      </Button>

   422→          <p className="text-xs text-muted-foreground py-2">No checklist items yet.</p>

  1023→          onChange={(e) => setSearchDraft(e.target.value)}

  1028→          <SelectTrigger className="h-9 sm:w-44"><SelectValue placeholder="All statuses" /></SelectTrigger>
  1029→          <SelectContent>

  1036→      {error ? (
  1037→        <ErrorState message="Projects could not be loaded." onRetry={() => void load()} />

  1059→          {filtered.map((p) => {

  1063→                key={p.id}
  1064→                onClick={() => { setDetailId(p.id); setDetailOpen(true); }}
  1065→                className="apex-panel p-4 text-left space-y-3 hover:border-primary/40 hover:shadow-lg hover:shadow-primary/5 transition-all focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-primary"
  1066→              >

  1102→      <ProjectDetailSheet
  1103→        projectId={detailId}
  1104→        open={detailOpen}
