    57→const num = (v: string): number => {
    58→  const n = parseFloat(v);
    59→  return Number.isFinite(n) ? n : 0;
    60→};
    61→const round2 = (n: number) => Math.round(n * 100) / 100;
    62→
    63→/** Mirrors the server formula: total = subtotal - discount + subtotal * tax% / 100 */
    64→function previewTotals(items: ItemDraft[], discount: number, taxPercent: number) {
    65→  const subtotal = round2(items.reduce((s, i) => s + num(i.quantity) * num(i.unitPrice), 0));
    66→  const tax = round2((subtotal * taxPercent) / 100);
    67→  return { subtotal, tax, total: round2(subtotal - discount + tax) };
    68→}

    70→function extractList(data: unknown): Record<string, unknown>[] {
    71→  if (Array.isArray(data)) return data as Record<string, unknown>[];
    72→  if (data && typeof data === "object") {
    73→    const obj = data as Record<string, unknown>;
    74→    for (const key of ["items", "leads", "clients", "projects", "team"]) {
    75→      if (Array.isArray(obj[key])) return obj[key] as Record<string, unknown>[];
    76→    }
    77→  }
    78→  return [];
    79→}

    87→function parseDeliverables(raw?: string | null): string[] {
    88→  if (!raw) return [];
    89→  try {
    90→    const v = JSON.parse(raw);
    91→    return Array.isArray(v) ? v.map(String) : [String(v)];
    92→  } catch {
    93→    return raw.split("\n").map((s) => s.trim()).filter(Boolean);
    94→  }

   410→function ProposalDetailSheet({ id, open, onOpenChange, onChanged, onConvert, canEdit, canDelete, onDeleteRequest }: {
   411→  id: string | null; open: boolean; onOpenChange: (o: boolean) => void;
   412→  onChanged: () => void; onConvert: (p: ProposalDetail) => void;
   413→  canEdit: boolean; canDelete: boolean; onDeleteRequest: (row: ProposalListRow) => void;
   414→}) {
   415→  const { toast } = useToast();
   416→  const [detail, setDetail] = useState<ProposalDetail | null>(null);
   417→  const [loading, setLoading] = useState(false);
   418→  const [acting, setActing] = useState(false);
   419→
   420→  const loadDetail = useCallback(async () => {
   421→    if (!id) return;
   422→    setLoading(true);
   423→    try {
   424→      setDetail(await api.get<ProposalDetail>(`/api/proposals/${id}`));
   425→    } catch (e) {
   426→      toast({ title: "Could not load proposal", description: e instanceof Error ? e.message : undefined });
   427→    } finally {
   428→      setLoading(false);
   429→    }
   430→  }, [id, toast]);
   431→
   432→  useEffect(() => {
   433→    if (open && id) loadDetail();
   434→    if (!open) setDetail(null);
   435→  }, [open, id, loadDetail]);
   436→
   437→  const act = async (status: string, successTitle: string) => {
   438→    if (!id) return;
   439→    setActing(true);
   440→    try {
   441→      await api.patch(`/api/proposals/${id}`, { status });
   442→      toast({ title: successTitle });

   867→                      {formatDate(r.validUntil)}{expired && " · expired"}
   868→                    </td>
   869→                    <td className="px-4 py-3 text-right">
   870→                      <div className="inline-flex items-center gap-1">
   871→                        <Button variant="ghost" size="icon" className="h-8 w-8" title="View" onClick={() => { setDetailId(r.id); setSheetOpen(true); }}>
   872→                          <Eye className="w-3.5 h-3.5" />
   873→                        </Button>
   874→                        {canEdit && !["ACCEPTED", "REJECTED"].includes(r.status) && (
   875→                          <Button variant="ghost" size="icon" className="h-8 w-8" title="Edit" onClick={() => openEdit(r)}>
   876→                            <Pencil className="w-3.5 h-3.5" />
   877→                          </Button>
   878→                        )}
   879→                        {canDelete && r.status === "DRAFT" && (
   880→                          <Button variant="ghost" size="icon" className="h-8 w-8 text-muted-foreground hover:text-destructive" title="Delete draft" onClick={() => setDeleting(r)}>
   881→                            <Trash2 className="w-3.5 h-3.5" />
   882→                          </Button>
   883→                        )}
   884→                        {!canEdit && !canDelete && (
   885→                          <DropdownMenu>
   886→                            <DropdownMenuTrigger asChild>
   887→                              <Button variant="ghost" size="icon" className="h-8 w-8"><MoreHorizontal className="w-4 h-4" /></Button>
   888→                            </DropdownMenuTrigger>
   889→                            <DropdownMenuContent align="end"><DropdownMenuItem disabled>No actions available</DropdownMenuItem></DropdownMenuContent>
   890→                          </DropdownMenu>
   891→                        )}
   892→                      </div>
   893→                    </td>
   894→                  </tr>
   895→                );
   896→              })}
   897→            </tbody>
   898→          </table>
   899→        </div>
   900→      )}
   901→
   902→      {totalPages > 1 && (
   903→        <div className="flex items-center justify-between text-sm mt-4">
   904→          <p className="text-muted-foreground">Page {page} of {totalPages} · {total} proposals</p>
   905→          <div className="flex gap-2">
   906→            <Button variant="outline" size="sm" disabled={page <= 1} onClick={() => load({ page: page - 1 })}>Previous</Button>
   907→            <Button variant="outline" size="sm" disabled={page >= totalPages} onClick={() => load({ page: page + 1 })}>Next</Button>
   908→          </div>
   909→        </div>
   910→      )}
   911→
   912→      {/* Builder dialog */}
   913→      <Dialog open={builderOpen} onOpenChange={setBuilderOpen}>
   914→        <DialogContent className="sm:max-w-3xl">
   915→          <DialogHeader>
   916→            <DialogTitle>{editingId ? "Edit proposal" : "New proposal"}</DialogTitle>
   917→            <DialogDescription>Totals are always recalculated on the server from the line items below.</DialogDescription>
   918→          </DialogHeader>
   919→          <div className="max-h-[70vh] overflow-y-auto apex-scroll pr-1 space-y-5">
   920→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   921→              <Field label="Title" required>
   922→                <Input value={form.title} onChange={(e) => setForm({ ...form, title: e.target.value })} placeholder="e.g. Nile Trading — E-commerce Platform" />
   923→              </Field>
   924→              <div className="grid grid-cols-2 gap-3">
   925→                <Field label="Currency">
   926→                  <Select value={form.currency} onValueChange={(v) => setForm({ ...form, currency: v })}>
   927→                    <SelectTrigger><SelectValue /></SelectTrigger>
   928→                    <SelectContent>
   929→                      <SelectItem value="EGP">EGP</SelectItem>
   930→                      <SelectItem value="USD">USD</SelectItem>
   931→                      <SelectItem value="SAR">SAR</SelectItem>
   932→                    </SelectContent>
   933→                  </Select>
   934→                </Field>
   935→                <Field label="Valid until">
   936→                  <Input type="date" value={form.validUntil} onChange={(e) => setForm({ ...form, validUntil: e.target.value })} />
   937→                </Field>
   938→              </div>
   939→            </div>
   940→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   941→              {leadOpts.length > 0 ? (
   942→                <Field label="Linked lead">
   943→                  <Select value={form.leadId || "NONE"} onValueChange={(v) => setForm({ ...form, leadId: v === "NONE" ? "" : v, clientId: "" })}>
   944→                    <SelectTrigger><SelectValue placeholder="None" /></SelectTrigger>
   945→                    <SelectContent className="max-h-60">
   946→                      <SelectItem value="NONE">None</SelectItem>
   947→                      {leadOpts.map((o) => <SelectItem key={o.id} value={o.id}>{o.label}</SelectItem>)}
   948→                    </SelectContent>
   949→                  </Select>
   950→                </Field>
   951→              ) : (
   952→                <Field label="Linked lead" hint="Lead list unavailable — the proposal will be unlinked">
   953→                  <Input value={form.leadId} disabled placeholder="Unavailable without leads.view" />
   954→                </Field>
   955→              )}
   956→              {clientOpts.length > 0 ? (
   957→                <Field label="Linked client">
   958→                  <Select value={form.clientId || "NONE"} onValueChange={(v) => setForm({ ...form, clientId: v === "NONE" ? "" : v, leadId: "" })}>
   959→                    <SelectTrigger><SelectValue placeholder="None" /></SelectTrigger>
   960→                    <SelectContent className="max-h-60">
   961→                      <SelectItem value="NONE">None</SelectItem>
   962→                      {clientOpts.map((o) => <SelectItem key={o.id} value={o.id}>{o.label}</SelectItem>)}
   963→                    </SelectContent>
   964→                  </Select>
   965→                </Field>
   966→              ) : (
   967→                <Field label="Linked client" hint="Client list unavailable — the proposal will be unlinked">
   968→                  <Input value={form.clientId} disabled placeholder="Unavailable without clients.view" />
   969→                </Field>
   970→              )}
   971→            </div>
   972→
   973→            <div>
   974→              <p className="text-xs font-medium text-muted-foreground mb-2">Line items</p>
   975→              <ItemsEditor items={items} onChange={setItems} currency={form.currency} />
   976→              <div className="mt-3 rounded-lg border border-border bg-secondary/30 p-4">
   977→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   978→                  <Field label="Discount amount">
   979→                    <Input type="number" min={0} step={0.01} value={form.discountAmount} onChange={(e) => setForm({ ...form, discountAmount: e.target.value })} />
   980→                  </Field>
   981→                  <Field label="Tax percent">
   982→                    <Input type="number" min={0} max={100} step={0.01} value={form.taxPercent} onChange={(e) => setForm({ ...form, taxPercent: e.target.value })} />
   983→                  </Field>
   984→                </div>
   985→                <Separator className="my-3" />
   986→                <div className="space-y-1.5 text-sm">
   987→                  <div className="flex justify-between"><span className="text-muted-foreground">Subtotal</span><span className="tabular-nums">{formatCurrency(totals.subtotal, form.currency)}</span></div>
   988→                  <div className="flex justify-between"><span className="text-muted-foreground">Tax ({num(form.taxPercent)}%)</span><span className="tabular-nums">{formatCurrency(totals.tax, form.currency)}</span></div>
   989→                  <div className="flex justify-between font-semibold"><span>Total</span><span className="tabular-nums text-primary">{formatCurrency(totals.total, form.currency)}</span></div>
   990→                </div>
   991→              </div>
   992→            </div>
   993→
   994→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   995→              <Field label="Problem"><Textarea rows={2} value={form.problem} onChange={(e) => setForm({ ...form, problem: e.target.value })} placeholder="What the client is struggling with" /></Field>
   996→              <Field label="Solution"><Textarea rows={2} value={form.solution} onChange={(e) => setForm({ ...form, solution: e.target.value })} placeholder="How APEX solves it" /></Field>
   997→            </div>
   998→            <Field label="Scope"><Textarea rows={2} value={form.scope} onChange={(e) => setForm({ ...form, scope: e.target.value })} placeholder="What is included and what is not" /></Field>
   999→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
  1000→              <Field label="Deliverables" hint="Comma or newline separated">
  1001→                <Textarea rows={2} value={form.deliverables} onChange={(e) => setForm({ ...form, deliverables: e.target.value })} placeholder="5 pages design, CMS setup, 1 month support" />
  1002→              </Field>
  1003→              <Field label="Timeline"><Textarea rows={2} value={form.timeline} onChange={(e) => setForm({ ...form, timeline: e.target.value })} placeholder="e.g. 4 weeks from kickoff" /></Field>
  1004→            </div>
  1005→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
  1006→              <Field label="Payment terms"><Textarea rows={2} value={form.paymentTerms} onChange={(e) => setForm({ ...form, paymentTerms: e.target.value })} placeholder="e.g. 50% upfront, 50% on delivery" /></Field>
  1007→              <Field label="Revision policy"><Textarea rows={2} value={form.revisionPolicy} onChange={(e) => setForm({ ...form, revisionPolicy: e.target.value })} placeholder="e.g. 2 revision rounds included" /></Field>
  1008→              <Field label="Maintenance terms"><Textarea rows={2} value={form.maintenanceTerms} onChange={(e) => setForm({ ...form, maintenanceTerms: e.target.value })} placeholder="e.g. 3 months free bug fixes" /></Field>
  1009→              <Field label="Terms"><Textarea rows={2} value={form.terms} onChange={(e) => setForm({ ...form, terms: e.target.value })} placeholder="Validity, ownership, confidentiality…" /></Field>
  1010→            </div>
  1011→            <Field label="Internal notes"><Textarea rows={2} value={form.notes} onChange={(e) => setForm({ ...form, notes: e.target.value })} placeholder="Notes the client will never see" /></Field>
  1012→          </div>
  1013→          <DialogFooter>
  1014→            <Button variant="outline" onClick={() => setBuilderOpen(false)}>Cancel</Button>
  1015→            <Button onClick={submitBuilder} disabled={saving}>
  1016→              {saving ? <RefreshCw className="w-4 h-4 mr-2 animate-spin" /> : null}
  1017→              {editingId ? "Save changes" : "Create proposal"}
  1018→            </Button>
  1019→          </DialogFooter>
  1020→        </DialogContent>
  1021→      </Dialog>
  1022→
  1023→      {/* Detail sheet */}
  1024→      <ProposalDetailSheet
  1025→        id={detailId}
  1026→        open={sheetOpen}
  1027→        onOpenChange={setSheetOpen}
  1028→        onChanged={() => load()}
  1029→        canEdit={canEdit}
  1030→        canDelete={canDelete}
  1031→        onDeleteRequest={(row) => setDeleting(row)}
  1032→        onConvert={(p) => { setConvertTarget(p); setConvertOpen(true); }}
  1033→      />
  1034→
  1035→      {/* Convert dialog */}
  1036→      <ConvertDialog
  1037→        proposal={convertTarget}
  1038→        open={convertOpen}
  1039→        onClose={() => { setConvertOpen(false); setConvertTarget(null); setSheetOpen(false); load(); }}
  1040→        onConverted={() => load()}
  1041→        navigate={navigate}
  1042→      />
  1043→
  1044→      {/* Delete confirm */}
  1045→      <AlertDialog open={!!deleting} onOpenChange={(o) => !o && setDeleting(null)}>
  1046→        <AlertDialogContent>
  1047→          <AlertDialogHeader>
  1048→            <AlertDialogTitle>Delete this draft?</AlertDialogTitle>
  1049→            <AlertDialogDescription>
  1050→              {deleting ? `${deleting.proposalNumber} — ${deleting.title}` : ""} will be permanently removed. Only drafts can be deleted.
  1051→            </AlertDialogDescription>
  1052→          </AlertDialogHeader>
  1053→          <AlertDialogFooter>
  1054→            <AlertDialogCancel>Keep draft</AlertDialogCancel>
  1055→            <AlertDialogAction className="bg-destructive text-destructive-foreground hover:bg-destructive/90" onClick={confirmDelete}>Delete</AlertDialogAction>
  1056→          </AlertDialogFooter>
  1057→        </AlertDialogContent>
  1058→      </AlertDialog>
  1059→    </div>
  1060→  );
  1061→}
