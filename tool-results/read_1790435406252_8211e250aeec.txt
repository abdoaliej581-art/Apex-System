   134→   134→function deadlineMeta(deadline?: string | null, status?: string): { text: string; cls: string } {
   135→   135→  if (!deadline) return { text: "No deadline", cls: "text-muted-foreground" };
   136→   136→  const d = new Date(deadline);
   137→   137→  d.setHours(0, 0, 0, 0);
   138→   138→  const today = new Date();
   139→   139→  today.setHours(0, 0, 0, 0);
   140→   140→  const days = Math.round((d.getTime() - today.getTime()) / 86400000);
   141→   141→  const urgent = !!status && !["COMPLETED", "CANCELLED"].includes(status) && days < 7;
   142→   142→  return { text: formatDate(deadline), cls: urgent ? "text-rose-300 font-medium" : "text-muted-foreground" };
   143→   143→}

   238→   238→        title: "Could not create project",
   239→   239→        description: e instanceof ApiClientError ? e.message : undefined,
   240→   240→        variant: "destructive",
   241→   241→      });

   244→   244→    }
