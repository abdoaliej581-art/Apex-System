    29→    29→/** Robustly extract a list from any list-route payload ({ items } | array | { leads }...) */
    30→    30→function extractList(data: unknown): Record<string, unknown>[] {
    31→    31→  if (Array.isArray(data)) return data as Record<string, unknown>[];
    32→    32→  if (data && typeof data === "object") {
    33→    33→    const obj = data as Record<string, unknown>;
    34→    34→    for (const key of ["items", "leads", "clients", "projects", "team"]) {
    35→    35→      if (Array.isArray(obj[key])) return obj[key] as Record<string, unknown>[];
    36→    36→    }
    37→    37→  }
    38→    38→  return [];
    39→    39→}
    40→    40→
    41→    41→function toOptions(url: string, labelKey: string): Promise<Option[]> {
    42→    42→  return api
    43→    43→    .get<unknown>(url)
    44→    44→    .then((data) =>
    45→    45→      extractList(data)
    46→    46→        .map((x) => ({ id: String(x.id), label: String(x[labelKey] ?? x.name ?? x.companyName ?? "") }))
    47→    47→        .filter((o) => o.label)
    48→    48→    )
    49→    49→    .catch(() => []); // 403 / unavailable → degrade gracefully to free text
    50→    50→}
    51→    51→
    52→    52→function toDateInput(d: string | Date): string {
    53→    53→  const x = new Date(d);
    54→    54→  const p = (n: number) => String(n).padStart(2, "0");
    55→    55→  return `${x.getFullYear()}-${p(x.getMonth() + 1)}-${p(x.getDate())}`;
    56→    56→}

   394→   394→                            <DropdownMenuTrigger asChild>
   395→   395→                              <Button variant="ghost" size="icon" className="h-8 w-8"><MoreHorizontal className="w-4 h-4" /></Button>
   396→   396→                            </DropdownMenuTrigger>
   397→   397→                            <DropdownMenuContent align="end"><DropdownMenuItem disabled>No actions available</DropdownMenuItem></DropdownMenuContent>
   398→   398→                          </DropdownMenu>
   399→   399→                        )}
   400→   400→                      </div>
   401→   401→                    </div>
   402→   402→                  </div>
   403→   403→                ))}
   404→   404→              </div>
   405→   405→            </div>
   406→   406→          ))}
   407→   407→
   408→   408→          {/* Pagination */}
   409→   409→          {totalPages > 1 && (
   410→   410→            <div className="flex items-center justify-between text-sm">
   411→   411→              <p className="text-muted-foreground">Page {page} of {totalPages} · {total} meetings</p>
   412→   412→              <div className="flex gap-2">
   413→   413→                <Button variant="outline" size="sm" disabled={page <= 1} onClick={() => load({ page: page - 1 })}>Previous</Button>
   414→   414→                <Button variant="outline" size="sm" disabled={page >= totalPages} onClick={() => load({ page: page + 1 })}>Next</Button>
   415→   415→              </div>
   416→   416→            </div>
   417→   417→          )}
   418→   418→        </div>
   419→   419→      )}
   420→   420→
   421→   421→      {/* Create / Edit dialog */}
   422→   422→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   423→   423→        <DialogContent className="sm:max-w-lg">
   424→   424→          <DialogHeader>
   425→   425→            <DialogTitle>{editing ? "Edit meeting" : "New meeting"}</DialogTitle>
   426→   426→            <DialogDescription>
   427→   427→              {editing ? "Update the schedule or details of this meeting." : "Link it to a lead, client or project to keep the timeline connected."}
   428→   428→            </DialogDescription>
   429→   429→          </DialogHeader>
   430→   430→          <div className="max-h-[65vh] overflow-y-auto apex-scroll pr-1 space-y-4">
   431→   431→            <Field label="Title" required>
   432→   432→              <Input value={form.title} onChange={(e) => setForm({ ...form, title: e.target.value })} placeholder="e.g. Requirements workshop with Nile" />
   433→   433→            </Field>
   434→   434→            <div className="grid grid-cols-3 gap-3">
   435→   435→              <Field label="Date" required>
   436→   436→                <Input type="date" value={form.date} onChange={(e) => setForm({ ...form, date: e.target.value })} />
   437→   437→              </Field>
   438→   438→              <Field label="Start" required>
   439→   439→                <Input type="time" value={form.startTime} onChange={(e) => setForm({ ...form, startTime: e.target.value })} />
   440→   440→              </Field>
   441→   441→              <Field label="End" required>
   442→   442→                <Input type="time" value={form.endTime} onChange={(e) => setForm({ ...form, endTime: e.target.value })} />
   443→   443→              </Field>
   444→   444→            </div>
   445→   445→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   446→   446→              {leadOpts.length > 0 ? (
   447→   447→                <Field label="Linked lead">
   448→   448→                  <Select value={form.leadId || "NONE"} onValueChange={(v) => setForm({ ...form, leadId: v === "NONE" ? "" : v, clientId: "" })}>
   449→   449→                    <SelectTrigger><SelectValue placeholder="None" /></SelectTrigger>
   450→   450→                    <SelectContent className="max-h-60">
   451→   451→                      <SelectItem value="NONE">None</SelectItem>
   452→   452→                      {leadOpts.map((o) => <SelectItem key={o.id} value={o.id}>{o.label}</SelectItem>)}
   453→   453→                    </SelectContent>
   454→   454→                  </Select>
   455→   455→                </Field>
   456→   456→              ) : (
   457→   457→                <Field label="Linked lead" hint={optsTried ? "Lead list unavailable — type a reference name instead" : undefined}>
   458→   458→                  <Input value={form.relatedName} onChange={(e) => setForm({ ...form, relatedName: e.target.value })} placeholder="Company / person (unlinked)" disabled={!!form.clientId} />
   459→   459→                </Field>
   460→   460→              )}
   461→   461→              {clientOpts.length > 0 ? (
   462→   462→                <Field label="Linked client">
   463→   463→                  <Select value={form.clientId || "NONE"} onValueChange={(v) => setForm({ ...form, clientId: v === "NONE" ? "" : v, leadId: "" })}>
   464→   464→                    <SelectTrigger><SelectValue placeholder="None" /></SelectTrigger>
   465→   465→                    <SelectContent className="max-h-60">
   466→   466→                      <SelectItem value="NONE">None</SelectItem>
   467→   467→                      {clientOpts.map((o) => <SelectItem key={o.id} value={o.id}>{o.label}</SelectItem>)}
   468→   468→                    </SelectContent>
   469→   469→                  </Select>
   470→   470→                </Field>
   471→   471→              ) : (
   472→   472→                <Field label="Linked client" hint={optsTried ? "Client list unavailable — type a reference name instead" : undefined}>
   473→   473→                  <Input value={form.relatedName} onChange={(e) => setForm({ ...form, relatedName: e.target.value })} placeholder="Company (unlinked)" disabled={!!form.leadId} />
   474→   474→                </Field>
   475→   475→              )}
   476→   476→            </div>
   477→   477→            {projectOpts.length > 0 && (
   478→   478→              <Field label="Linked project">
   479→   479→                <Select value={form.projectId || "NONE"} onValueChange={(v) => setForm({ ...form, projectId: v === "NONE" ? "" : v })}>
   480→   480→                  <SelectTrigger><SelectValue placeholder="None" /></SelectTrigger>
   481→   481→                  <SelectContent className="max-h-60">
   482→   482→                    <SelectItem value="NONE">None</SelectItem>
   483→   483→                    {projectOpts.map((o) => <SelectItem key={o.id} value={o.id}>{o.label}</SelectItem>)}
   484→   484→                  </SelectContent>
   485→   485→                </Select>
   486→   486→              </Field>
   487→   487→            )}
   488→   488→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   489→   489→              <Field label="Location">
   490→   490→                <Input value={form.location} onChange={(e) => setForm({ ...form, location: e.target.value })} placeholder="Office / client site" />
   491→   491→              </Field>
   492→   492→              <Field label="Meeting link">
   493→   493→                <Input value={form.meetingLink} onChange={(e) => setForm({ ...form, meetingLink: e.target.value })} placeholder="https://meet.google.com/…" />
   494→   494→              </Field>
   495→   495→            </div>
   496→   496→            <Field label="Notes">
   497→   497→              <Textarea value={form.notes} onChange={(e) => setForm({ ...form, notes: e.target.value })} rows={3} placeholder="Agenda, attendees, preparation…" />
   498→   498→            </Field>
   499→   499→          </div>
   500→   500→          <DialogFooter>
   501→   501→            <Button variant="outline" onClick={() => setDialogOpen(false)}>Cancel</Button>
   502→   502→            <Button onClick={submit} disabled={saving}>{saving ? "Saving…" : editing ? "Save changes" : "Schedule meeting"}</Button>
   503→   503→          </DialogFooter>
   504→   504→        </DialogContent>
   505→   505→      </Dialog>
   506→   506→
   507→   507→      {/* Complete dialog (outcome) */}
   508→   508→      <Dialog open={!!completing} onOpenChange={(o) => !o && setCompleting(null)}>
   509→   509→        <DialogContent className="sm:max-w-md">
   510→   510→          <DialogHeader>
   511→   511→            <DialogTitle>Complete meeting</DialogTitle>
   512→   512→            <DialogDescription>Record what happened and what comes next.</DialogDescription>
   513→   513→          </DialogHeader>
   514→   514→          <div className="space-y-4">
   515→   515→            <Field label="Outcome">
   516→   516→              <Textarea value={outcome} onChange={(e) => setOutcome(e.target.value)} rows={3} placeholder="e.g. Client approved scope, wants the proposal by Sunday." />
   517→   517→            </Field>
   518→   518→            <Field label="Next action">
   519→   519→              <Input value={nextAction} onChange={(e) => setNextAction(e.target.value)} placeholder="e.g. Send proposal before Thursday" />
   520→   520→            </Field>
   521→   521→          </div>
   522→   522→          <DialogFooter>
   523→   523→            <Button variant="outline" onClick={() => setCompleting(null)}>Cancel</Button>
   524→   524→            <Button onClick={submitComplete}>Mark completed</Button>
   525→   525→          </DialogFooter>
   526→   526→        </DialogContent>
   527→   527→      </Dialog>
   528→   528→
   529→   529→      {/* Delete confirm */}
   530→   530→      <AlertDialog open={!!deleting} onOpenChange={(o) => !o && setDeleting(null)}>
   531→   531→        <AlertDialogContent>
   532→   532→          <AlertDialogHeader>
   533→   533→            <AlertDialogTitle>Delete this meeting?</AlertDialogTitle>
   534→   534→            <AlertDialogDescription>
   535→   535→              &quot;{deleting?.title}&quot; will be permanently removed. This cannot be undone.
   536→   536→            </AlertDialogDescription>
   537→   537→          </AlertDialogHeader>
   538→   538→          <AlertDialogFooter>
   539→   539→            <AlertDialogCancel>Keep meeting</AlertDialogCancel>
   540→   540→            <AlertDialogAction className={cn("bg-destructive text-destructive-foreground hover:bg-destructive/90")} onClick={confirmDelete}>
   541→   541→              Delete
   542→   542→            </AlertDialogAction>
   543→   543→          </AlertDialogFooter>
   544→   544→        </AlertDialogContent>
   545→   545→      </AlertDialog>
   546→   546→    </div>
   547→   547→  );
   548→   548→}
