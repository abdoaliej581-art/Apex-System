   117→/** Date → value usable by <input type="datetime-local"> in the user's local timezone. */
   118→export function toLocalInput(d: string | Date | null | undefined): string {
   119→  if (!d) return "";
   120→  const dt = new Date(d);
   121→  if (Number.isNaN(dt.getTime())) return "";
   122→  const pad = (n: number) => String(n).padStart(2, "0");
   123→  return `${dt.getFullYear()}-${pad(dt.getMonth() + 1)}-${pad(dt.getDate())}T${pad(dt.getHours())}:${pad(dt.getMinutes())}`;
   124→}
   125→
   126→export function isOverdueDate(d: string | Date | null | undefined): boolean {
   127→  if (!d) return false;
   128→  const t = new Date(d).getTime();
   129→  return !Number.isNaN(t) && t < Date.now();
   130→}

   132→/** Fetch active team members; returns empty list when the caller lacks team.view. */
   133→export function useTeam(): { team: TeamMember[]; hasAccess: boolean } {
   134→  const [team, setTeam] = useState<TeamMember[]>([]);
   135→  const [hasAccess, setHasAccess] = useState(false);
   136→
   137→  useEffect(() => {
   138→    let cancelled = false;
   139→    api.get<{ team: TeamMember[] }>("/api/team")
   140→      .then((d) => { if (!cancelled) { setTeam(d.team); setHasAccess(true); } })
   141→      .catch(() => { if (!cancelled) { setTeam([]); setHasAccess(false); } });
   142→    return () => { cancelled = true; };
   143→  }, []);
   144→
   145→  return { team, hasAccess };
   146→}

   292→export function LeadFormDialog({
   293→  open, onOpenChange, lead, onSaved,
   294→}: {
   295→  open: boolean;
   296→  onOpenChange: (o: boolean) => void;
   297→  lead?: Lead | null;
   298→  onSaved: () => void;
   299→}) {
   300→  const { toast } = useToast();
   301→  const { team, hasAccess } = useTeam();
   302→  const [values, setValues] = useState<LeadFormValues>(emptyLeadForm);
   303→  const [errors, setErrors] = useState<Record<string, string>>({});
   304→  const [saving, setSaving] = useState(false);
   305→  const isEdit = Boolean(lead);
   306→
   307→  useEffect(() => {
   308→    if (open) {
   309→      setValues(lead ? leadToForm(lead) : emptyLeadForm);
   310→      setErrors({});
   311→    }
   312→  }, [open, lead]);
   313→
   314→  const set = (key: keyof LeadFormValues, value: string) =>
   315→    setValues((v) => ({ ...v, [key]: value }));
   316→
   317→  const validate = (): boolean => {
   318→    const e: Record<string, string> = {};
   319→    if (values.companyName.trim().length < 2) e.companyName = "Company name is required (min 2 characters).";
   320→    if (values.contactName.trim().length < 2) e.contactName = "Contact name is required (min 2 characters).";
   321→    if (values.phone.trim().length < 7) e.phone = "Phone number is required (min 7 characters).";
   322→    if (values.estimatedBudget && Number.isNaN(Number(values.estimatedBudget))) e.estimatedBudget = "Budget must be a number.";
   323→    setErrors(e);
   324→    return Object.keys(e).length === 0;
   325→  };
   326→
   327→  const submit = async () => {
   328→    if (!validate()) return;
   329→    setSaving(true);
   330→    const payload = {
   331→      companyName: values.companyName.trim(),
   332→      contactName: values.contactName.trim(),
   333→      phone: values.phone.trim(),
   334→      email: values.email.trim(),
   335→      website: values.website.trim(),
   336→      instagram: values.instagram.trim(),
   337→      facebook: values.facebook.trim(),
   338→      linkedin: values.linkedin.trim(),
   339→      industry: values.industry.trim(),
   340→      location: values.location.trim(),
   341→      source: values.source,
   342→      serviceInterest: values.serviceInterest.trim(),
   343→      estimatedBudget: values.estimatedBudget === "" ? null : Number(values.estimatedBudget),
   344→      priority: values.priority,
   345→      assignedToId: values.assignedToId || null,
   346→      nextFollowUpAt: values.nextFollowUpAt ? new Date(values.nextFollowUpAt).toISOString() : null,
   347→      notes: values.notes.trim() || null,
   348→    };
   349→    try {
   350→      if (isEdit && lead) {
   351→        await api.patch<Lead>(`/api/leads/${lead.id}`, payload);
   352→        toast({ title: "Lead updated", description: `${payload.companyName} was saved successfully.` });
   353→      } else {
   354→        const created = await api.post<Lead>("/api/leads", payload);
   355→        toast({ title: "Lead created", description: `${created.leadNumber} — ${created.companyName}` });
   356→      }
   357→      onOpenChange(false);
   358→      onSaved();
   359→    } catch (err) {
   360→      toast({ title: isEdit ? "Update failed" : "Create failed", description: err instanceof Error ? err.message : "Something went wrong.", variant: "destructive" });
   361→    } finally {
   362→      setSaving(false);
   363→    }
   364→  };
   365→
   366→  return (
   367→    <Dialog open={open} onOpenChange={onOpenChange}>
   368→      <DialogContent className="sm:max-w-2xl max-h-[92vh] overflow-y-auto apex-scroll">
   369→        <DialogHeader>
   370→          <DialogTitle>{isEdit ? `Edit lead — ${lead?.companyName}` : "New lead"}</DialogTitle>
   371→          <DialogDescription>
   372→            {isEdit ? "Update the lead details below." : "Capture a new sales opportunity. Fields marked * are required."}
   373→          </DialogDescription>
   374→        </DialogHeader>
   375→
   376→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   377→          <Field label="Company name" required>
   378→            <Input value={values.companyName} onChange={(e) => set("companyName", e.target.value)} placeholder="Acme Ltd." />
   379→            {errors.companyName && <p className="text-xs text-destructive">{errors.companyName}</p>}
   380→          </Field>
   381→          <Field label="Contact person" required>
   382→            <Input value={values.contactName} onChange={(e) => set("contactName", e.target.value)} placeholder="John Smith" />
   383→            {errors.contactName && <p className="text-xs text-destructive">{errors.contactName}</p>}
   384→          </Field>
   385→          <Field label="Phone" required>
   386→            <Input value={values.phone} onChange={(e) => set("phone", e.target.value)} placeholder="+20 100 000 0000" dir="ltr" />
   387→            {errors.phone && <p className="text-xs text-destructive">{errors.phone}</p>}
   388→          </Field>
   389→          <Field label="Email">
   390→            <Input value={values.email} onChange={(e) => set("email", e.target.value)} placeholder="hello@acme.com" type="email" dir="ltr" />
   391→          </Field>
   392→          <Field label="Website">
   393→            <Input value={values.website} onChange={(e) => set("website", e.target.value)} placeholder="https://acme.com" dir="ltr" />
   394→          </Field>
   395→          <Field label="Industry">
   396→            <Input value={values.industry} onChange={(e) => set("industry", e.target.value)} placeholder="Real estate, F&B…" />
   397→          </Field>
   398→          <Field label="Instagram">
   399→            <Input value={values.instagram} onChange={(e) => set("instagram", e.target.value)} placeholder="@acme" dir="ltr" />
   400→          </Field>
   401→          <Field label="Facebook">

   766→export function LeadDetailDrawer({

   770→  open: boolean;

   772→  onChanged: () => void;
   773→  navigate: (p: string) => void;
   774→  permissions: string[];
   775→}) {
   776→  const { toast } = useToast();
   777→  const [data, setData] = useState<LeadDetailResponse | null>(null);
   778→  const [loading, setLoading] = useState(false);
   779→  const [error, setError] = useState<string | null>(null);
   780→  const [editOpen, setEditOpen] = useState(false);
   781→  const [convertOpen, setConvertOpen] = useState(false);
   782→  const [lostOpen, setLostOpen] = useState(false);
   783→  const [busy, setBusy] = useState(false);
   784→
   785→  const canEdit = permissions.includes("leads.edit");
   786→  const canConvert = permissions.includes("clients.create");
   787→
   788→  const load = useCallback(async () => {
   789→    if (!leadId) return;
   790→    setLoading(true);
   791→    setError(null);
   792→    try {
   793→      setData(await api.get<LeadDetailResponse>(`/api/leads/${leadId}`));
   794→    } catch (err) {
   795→      setError(err instanceof Error ? err.message : "Failed to load this lead.");
   796→    } finally {
   797→      setLoading(false);
   798→    }
   799→  }, [leadId]);
   800→
   801→  useEffect(() => {
   802→    if (open && leadId) load();
   803→    if (!open) setData(null);

  1074→/** Assignee select used inside the drawer (loads team on mount). */

  1076→  currentId, onChange, disabled,
  1077→}: {
  1078→  currentId: string | null;
  1079→  onChange: (id: string | null) => void;
  1080→  disabled?: boolean;

  1086→  }

  1088→    <Select value={value} onValueChange={(v) => onChange(v === "UNASSIGNED" ? null : v)} disabled={disabled}>
  1089→      <SelectTrigger><SelectValue /></SelectTrigger>
  1090→      <SelectContent>
  1091→        <SelectItem value="UNASSIGNED">Unassigned</SelectItem>
  1092→        {team.map((m) => <SelectItem key={m.id} value={m.id}>{m.name}</SelectItem>)}
  1093→      </SelectContent>
  1094→    </Select>
  1095→  );
